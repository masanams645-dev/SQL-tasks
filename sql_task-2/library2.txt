CREATE OR REPLACE FUNCTION GetMemberBorrowDetails(p_member_id INT)
RETURNS TABLE (
    book_name TEXT,
    borrow_date DATE
)
LANGUAGE SQL
AS $$
    SELECT b.book_name::TEXT,
           br.borrow_date
    FROM Books b
    JOIN Borrow br
    ON b.book_id = br.book_id
    WHERE br.member_id = p_member_id;
$$;
